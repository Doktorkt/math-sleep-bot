@bot.message_handler(commands=["stats"])
def show_stats(message):
    if not os.path.exists(HISTORY_FILE):
        bot.send_message(message.chat.id, "📭 Дневник пуст.")
        return
    with open(HISTORY_FILE, "r", encoding="utf-8") as f:
        lines = f.readlines()
    durations = []
    for l in lines[-7:]:
        match = re.search(r"Сон:\s*([\d.]+)", l)
        if match:
            durations.append(float(match.group(1)))
    if durations:
        avg = sum(durations) / len(durations)
        bot.send_message(message.chat.id, "📊 Средний сон за неделю: `" + f"{avg:.1f}" + " ч.`", parse_mode="Markdown")
    else:
        bot.send_message(message.chat.id, "📭 Данных нет.")

@bot.message_handler(func=lambda m: m.chat.id in user_data and "sleep" not in user_data[m.chat.id])

    try:
        h, mt = map(int, m.text.split(":"))
        user_data[m.chat.id] = {"sleep_h": h, "sleep_m": mt, "sleep": h + (mt / 60.0)}
        bot.send_message(m.chat.id, "⏳ Принято! А во сколько ты просыпаешься? Выбери кнопку:", reply_markup=get_wake_keyboard())

@bot.message_handler(func=lambda m: m.chat.id in user_data and "sleep" in user_data[m.chat.id])
def handle_wake(m):
    try:
        h, mt = map(int, m.text.split(":"))
        x2 = h + (mt / 60.0)
        x1 = user_data[m.chat.id]["sleep"]
        
        duration = x2 - x1 if x2 >= x1 else (24 - x1) + x2
        energy = max(0.0, min(100.0, 100.0 - (abs(duration - 8.0) * 10) if 7.5 <= duration <= 8.5 else (65.0 + ((duration - 5.5) * 15) if 5.5 <= duration < 7.5 else (max(50.0, 100.0 - (duration - 8.5) * 20) if duration > 8.5 else (duration / 5.5) * 60.0))))
        tips = "Идеально! 🚀" if energy >= 85 else ("Нормально 👍" if energy >= 65 else ("Вялость ☕" if energy >= 40 else "Зомби 😴"))
        
        today = datetime.datetime.now().strftime("%d.%m.%Y")
        with open(HISTORY_FILE, "a", encoding="utf-8") as f_out:
            f_out.write("📅 *" + today + "* — Сон: `" + str(round(duration, 1)) + " ч.` | Бодрость: `" + str(round(energy, 1)) + "%`\n")
            
        sh = user_data[m.chat.id]["sleep_h"]
        sm = user_data[m.chat.id]["sleep_m"]
        
        cycles = []
        for c in range(2, 7):
            total_minutes = sh * 60 + sm + 15 + c * 90
            c_h = (total_minutes // 60) % 24
            c_m = total_minutes % 60
            str_h = "0" + str(c_h) if c_h < 10 else str(c_h)
            str_m = "0" + str(c_m) if c_m < 10 else str(c_m)
            cycles.append("⏰ " + str_h + ":" + str_m + " (" + str(round((c * 90) / 60, 1)) + " ч.)")
            
        response = (
            "📊 *Расчет сна:*\n"
            "⏳ Время: " + str(round(duration, 1)) + " ч.\n"
            "⚡ Бодрость: " + str(round(energy, 1)) + "%\n"
            "💡 Совет: " + tips + "\n\n"
            "💾 Дневник обновлен: /history\n"
            "📊 Неделя: /stats\n\n"
            "💤 *Фазы сна:* \n" + "\n".join(cycles) + "\n\n"
            "📖 *А пока малыш засыпает, почитайте ему сказку на ночь в вашем Сообществе!*\n"
            "Все рассказы и волшебные сказки ждут вас на канале:\n"
            "👉 t.me/dima_stories\n\n"
            "Нажми /start для нового расчета."
        )
        bot.send_message(m.chat.id, response, reply_markup=types.ReplyKeyboardRemove())
        del user_data[m.chat.id]
    except Exception as e:
        bot.send_message(m.chat.id, "❌ Ошибка расчета: " + str(e))
